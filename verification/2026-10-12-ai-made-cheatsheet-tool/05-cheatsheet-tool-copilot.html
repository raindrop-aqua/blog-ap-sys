## PostgreSQL 保守担当者向けチートシート

### 接続

#### ローカル接続
psql -U postgres

#### データベースを指定して接続
psql -U postgres -d mydb

#### リモート接続
psql -h 192.168.1.100 -p 5432 -U postgres -d mydb

#### 接続文字列形式
psql "host=192.168.1.100 port=5432 dbname=mydb user=postgres"

#### パスワード指定（確認用）
export PGPASSWORD='password'
psql -h localhost -U postgres -d mydb

## psqlメタコマンド

\conninfo

## 基本情報確認SQL

### PostgreSQLバージョン
SELECT version();

### 現在のDB名
SELECT current_database();

### 現在のユーザー
SELECT current_user;

### 現在時刻
SELECT now();

## データ確認

### 件数確認
SELECT COUNT(*) FROM users;

### 先頭100件表示
SELECT *
FROM users
LIMIT 100;

### テーブル構造確認
SELECT *
FROM information_schema.columns
WHERE table_name = 'users';

## セッション・接続確認

### 現在の接続一覧
SELECT *
FROM pg_stat_activity;

### 必要最低限の確認
SELECT
    pid,
    usename,
    application_name,
    client_addr,
    state,
    query
FROM pg_stat_activity;

## ロック確認

### ロック状況確認
SELECT *
FROM pg_locks;

### ロック待ち確認
SELECT
    pid,
    usename,
    wait_event_type,
    wait_event,
    query
FROM pg_stat_activity
WHERE wait_event IS NOT NULL;

## サイズ確認

### テーブルサイズ
SELECT
    pg_size_pretty(
        pg_relation_size('users')
    );

### テーブル総サイズ（インデックス含む）
SELECT
    pg_size_pretty(
        pg_total_relation_size('users')
    );

### サイズの大きいテーブル一覧
SELECT
    relname,
    pg_size_pretty(
        pg_total_relation_size(relid)
    ) AS total_size
FROM pg_catalog.pg_statio_user_tables
ORDER BY pg_total_relation_size(relid) DESC;

## インデックス確認

### テーブルのインデックス確認
\d users

### SQLで確認
SELECT
    indexname,
    indexdef
FROM pg_indexes
WHERE tablename = 'users';

## SQL性能調査

### 実行計画確認
EXPLAIN
SELECT *
FROM users
WHERE id = 100;

### 実行時間付き実行計画
EXPLAIN ANALYZE
SELECT *
FROM users
WHERE id = 100;

## メンテナンス

### 統計情報更新
ANALYZE users;

### 不要領域回収
VACUUM users;

### 両方実施
VACUUM ANALYZE users;

### 全データベース対象
VACUUM ANALYZE;

## 障害時によく使うコマンド

### 長時間実行中のSQL確認
SELECT
    pid,
    now() - query_start AS duration,
    usename,
    state,
    query
FROM pg_stat_activity
WHERE state <> 'idle'
ORDER BY duration DESC;

### セッション終了
SELECT pg_terminate_backend(PID);
例 SELECT pg_terminate_backend(12345);
※ 本番環境では影響確認必須
